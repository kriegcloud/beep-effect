import assert from 'node:assert/strict';
import { XMLParser } from 'fast-xml-parser';
import { COMMON_HTML, CURRENCY, ENTITY_ACTION, EntityDecoder } from '@nodable/entities';
import { HTML, XML, isUnsafe } from 'is-unsafe';

const options = { ignoreAttributes: false, attributeNamePrefix: '', parseAttributeValue: false,
  parseTagValue: false, textNodeName: '#text', trimValues: false };
const legacy = new XMLParser({ ...options, htmlEntities: true });
const replacement = () => new XMLParser({ ...options, entityDecoder: new EntityDecoder({
  namedEntities: { ...COMMON_HTML, ...CURRENCY }, numericAllowed: true,
  limit: { maxTotalExpansions: Infinity, maxExpandedLength: 100000, applyLimitsTo: 'all' },
  onInputEntity: (_name, value) => isUnsafe(value, [HTML, XML]) ? ENTITY_ACTION.BLOCK : ENTITY_ACTION.ALLOW,
}) });
const inputs = [
  '<r>A&nbsp;B&euro;&#39;&#x41;&amp;</r>',
  '<r>&unknown;</r>',
  '<r>We&amp;#39;re &lt;b&gt;here&lt;/b&gt;</r>',
  '<!DOCTYPE r [<!ENTITY speaker "We">]><r>&speaker;&apos;re</r>',
  '<!DOCTYPE r [<!ENTITY speaker "&lt;script&gt;alert(1)&lt;/script&gt;">]><r>&speaker;</r>',
  '<r>' + '&euro;'.repeat(100001) + '</r>',
  '<r>after-limit-reset&nbsp;&#x41;</r>',
  '<?xml version="1.1"?><r>&#x1F;</r>',
  '<r>&#x1F;</r>',
];
const parse = (parser, text) => {
  try { return { value: parser.parse(text) }; }
  catch (error) { return { error: error.message }; }
};
for (const text of inputs) assert.deepEqual(parse(replacement(), text), parse(legacy, text));
console.log('PASS: nine legacy/replacement cases, including expansion limits and XML-version isolation.');
